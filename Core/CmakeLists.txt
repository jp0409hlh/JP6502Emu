add_library(Core
    Source/Application.cpp
    Source/Window.cpp
)

    Source/AppErrorType.h
    Source/Application.h
    Source/ImguiWindow.h
    Source/Window.h
)

    PUBLIC 
        ${CMAKE_CURRENT_SOURCE_DIR}/Source
)

    PUBLIC
        imgui
        SDL3::SDL3
)
